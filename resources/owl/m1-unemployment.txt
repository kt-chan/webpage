    rdfs:label "Connected Master 7-Stage SOP Ontology with Connected Domain Entities" ;
    rdfs:comment "Fixed ontology seamlessly linking Debtor, Debt Account, Financial Health Profile, and SOP Dialogue Execution." .

#################################################################
#    Classes
#################################################################

### Actors & Domain Entities
m1sop:Party rdf:type owl:Class ;
    rdfs:label "Party" .

m1sop:Debtor rdf:type owl:Class ;
    rdfs:subClassOf m1sop:Party ;
    rdfs:label "Debtor" .

m1sop:CollectionAgent rdf:type owl:Class ;
    rdfs:subClassOf m1sop:Party ;
    rdfs:label "Collection Agent" .

m1sop:DebtAccount rdf:type owl:Class ;
    rdfs:label "Debt Account" .

m1sop:FinancialHealthProfile rdf:type owl:Class ;
    rdfs:label "Financial Health Profile" ;
    rdfs:comment "Captures income streams, unemployment status, UI benefits, and liquid assets." .

m1sop:ReasonForDelinquency rdf:type owl:Class ;
    rdfs:label "Reason For Delinquency (RFD)" .

### Interaction & SOP Execution
m1sop:Call rdf:type owl:Class ;
    rdfs:label "Call Interaction Session" .

m1sop:MasterSOPStage rdf:type owl:Class ;
    rdfs:label "Master SOP Stage" .

m1sop:SubStage rdf:type owl:Class ;
    rdfs:label "Sub-Stage / Branch Stage" .

m1sop:ComplianceArtifact rdf:type owl:Class ;
    rdfs:label "Compliance Artifact" .

    rdfs:label "Call Outcome" .

m1sop:ForbearanceAgreement rdf:type owl:Class ;

#################################################################
#    Object Properties (Fixing Entity Connections)
#################################################################

### 1. Connecting Debtor <-> DebtAccount
m1sop:holdsAccount rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:Debtor ;
    rdfs:range m1sop:DebtAccount ;
    owl:inverseOf m1sop:accountHeldBy ;
    rdfs:comment "Links a debtor to their delinquent credit card account." .

m1sop:accountHeldBy rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:DebtAccount ;
    rdfs:range m1sop:Debtor .

### 2. Connecting Debtor <-> FinancialHealthProfile
m1sop:hasFinancialProfile rdf:type owl:ObjectProperty , owl:FunctionalProperty ;
    rdfs:domain m1sop:Debtor ;
    rdfs:range m1sop:FinancialHealthProfile ;
    owl:inverseOf m1sop:profileOfDebtor ;
    rdfs:comment "Links a debtor to their active financial assessment." .

m1sop:profileOfDebtor rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:FinancialHealthProfile ;
    rdfs:range m1sop:Debtor .

### 3. Connecting DebtAccount <-> FinancialHealthProfile
m1sop:associatedFinancialProfile rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:DebtAccount ;
    rdfs:range m1sop:FinancialHealthProfile ;
    rdfs:comment "Associates the financial profile directly to the specific debt account under review." .

### 4. Connecting Call Session to Entities
m1sop:hasDebtorParticipant rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:Call ;
    rdfs:range m1sop:Debtor .

m1sop:concernsAccount rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:Call ;
    rdfs:range m1sop:DebtAccount .

m1sop:conductedByAgent rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:Call ;
    rdfs:range m1sop:CollectionAgent .

m1sop:evaluatesFinancialProfile rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:Call ;
    rdfs:range m1sop:FinancialHealthProfile ;
    rdfs:comment "Captures or updates the financial health profile during Stage 3 profiling." .

### 5. Dialogue Execution & SOP Properties
m1sop:currentMasterStage rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:Call ;
    rdfs:range m1sop:MasterSOPStage .

m1sop:currentSubStage rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:Call ;
    rdfs:range m1sop:SubStage .

m1sop:detectedRFD rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:Call ;
    rdfs:range m1sop:ReasonForDelinquency .

m1sop:nextMasterStage rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:MasterSOPStage ;
    rdfs:range m1sop:MasterSOPStage .

m1sop:branchesToSubStage rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:MasterSOPStage ;
    rdfs:range m1sop:SubStage .

m1sop:nextSubStage rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:SubStage ;
    rdfs:range m1sop:SubStage .

m1sop:rejoinsMasterStage rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:SubStage ;
    rdfs:range m1sop:MasterSOPStage .

m1sop:requiresCompliance rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:MasterSOPStage ;
    rdfs:range m1sop:ComplianceArtifact .

m1sop:loggedCompliance rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:Call ;
    rdfs:range m1sop:ComplianceArtifact .

m1sop:yieldedOutcome rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:Call ;
    rdfs:range m1sop:CallOutcome .

#################################################################
#    Fully Connected Runtime Execution Example
#################################################################

# 1. Debtor
m1sop:Debtor_10492 a m1sop:Debtor ;
    rdfs:label "John Doe" ;
    m1sop:holdsAccount m1sop:Account_88201 ;
    m1sop:hasFinancialProfile m1sop:FinancialProfile_10492 .

# 2. Debt Account
m1sop:Account_88201 a m1sop:DebtAccount ;
    m1sop:outstandingBalance 1850.00 ;
    m1sop:openToBuyCredit 3150.00 ;
    m1sop:associatedFinancialProfile m1sop:FinancialProfile_10492 .

# 3. Financial Health Profile (Now explicitly bound to Debtor and Account)
m1sop:FinancialProfile_10492 a m1sop:FinancialHealthProfile ;
    m1sop:isReceivingUIBenefits "false"^^xsd:boolean ;
    m1sop:expectedUIStartReceiptDate "2026-10-22"^^xsd:date ;
    m1sop:monthlySeveranceIncome 0.00 .

# 4. Live Call Session (Traverses and evaluates all connected nodes)
m1sop:ActiveCall_9901 a m1sop:Call ;
    m1sop:hasDebtorParticipant m1sop:Debtor_10492 ;
    m1sop:concernsAccount m1sop:Account_88201 ;
    m1sop:evaluatesFinancialProfile m1sop:FinancialProfile_10492 ;
    m1sop:conductedByAgent m1sop:Agent_Bot_01 ;
    m1sop:detectedRFD m1sop:RFD_InvoluntaryUnemployment ;
    m1sop:currentMasterStage m1sop:Stage_3_Inquiry ;
    m1sop:currentSubStage m1sop:SubStage_3_2_FinancialProfiling ;
    m1sop:loggedCompliance m1sop:Req_Verification , m1sop:Req_MiniMiranda .
