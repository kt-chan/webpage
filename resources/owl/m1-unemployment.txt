@prefix m1sop: <http://example.org/ontology/m1-unemployment-sop#> .
@prefix owl:   <http://www.w3.org/2002/07/owl#> .
@prefix rdf:   <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .

@prefix xsd:   <http://www.w3.org/2001/XMLSchema#> .
@prefix rdfs:  <http://www.w3.org/2000/01/rdf-schema#> .

<http://example.org/ontology/m1-unemployment-sop> rdf:type owl:Ontology ;
    rdfs:label "Fully Connected M1-Unemployment SOP Domain & Workflow Ontology" ;
    rdfs:comment "Fixes entity isolation by explicitly linking Party/Debtor, DebtAccount, FinancialHealthProfile, and Call interactions." .

#################################################################
#    Classes
#################################################################

m1sop:Party rdf:type owl:Class ;
    rdfs:label "Party" .

m1sop:Debtor rdf:type owl:Class ;
    rdfs:subClassOf m1sop:Party ;
    rdfs:label "Debtor" .

m1sop:CollectionAgent rdf:type owl:Class ;
    rdfs:subClassOf m1sop:Party ;
    rdfs:label "Collection Agent" .

m1sop:DebtAccount rdf:type owl:Class ;
    rdfs:label "Debt Account" .

m1sop:FinancialHealthProfile rdf:type owl:Class ;
    rdfs:label "Financial Health Profile" .

m1sop:Call rdf:type owl:Class ;
    rdfs:label "Call Session" .

m1sop:MasterSOPStage rdf:type owl:Class ;
    rdfs:label "Master SOP Stage" .

m1sop:SubStage rdf:type owl:Class ;
    rdfs:label "Sub-Stage" .

m1sop:ReasonForDelinquency rdf:type owl:Class ;
    rdfs:label "Reason For Delinquency (RFD)" .

m1sop:CallOutcome rdf:type owl:Class ;
    rdfs:label "Call Outcome" .

m1sop:ComplianceArtifact rdf:type owl:Class ;
    rdfs:label "Compliance Artifact" .

#################################################################
#    Object Properties: Connecting Domain Entities
#################################################################

### 1. Party / Debtor <-> DebtAccount Relationships
m1sop:holdsAccount rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:Party ;
    rdfs:range m1sop:DebtAccount ;
    rdfs:label "holds account" .

m1sop:accountHeldBy rdf:type owl:ObjectProperty ;
    owl:inverseOf m1sop:holdsAccount ;
    rdfs:domain m1sop:DebtAccount ;
    rdfs:range m1sop:Party ;
    rdfs:label "account held by" .

### 2. Party / Debtor / Account <-> FinancialHealthProfile Relationships
m1sop:hasFinancialProfile rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:Debtor ;
    rdfs:range m1sop:FinancialHealthProfile ;
    rdfs:label "has financial profile" .

m1sop:profileBelongsTo rdf:type owl:ObjectProperty ;
    owl:inverseOf m1sop:hasFinancialProfile ;
    rdfs:domain m1sop:FinancialHealthProfile ;
    rdfs:range m1sop:Debtor ;
    rdfs:label "profile belongs to" .

m1sop:accountFinancialProfile rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:DebtAccount ;
    rdfs:range m1sop:FinancialHealthProfile ;
    rdfs:label "account financial profile" .

### 3. Call Session <-> Domain Entity Traversal
m1sop:hasDebtorParticipant rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:Call ;
    rdfs:range m1sop:Debtor ;
    rdfs:label "has debtor participant" .

m1sop:concernsAccount rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:Call ;
    rdfs:range m1sop:DebtAccount ;
    rdfs:label "concerns account" .

m1sop:evaluatesFinancialProfile rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:Call ;
    rdfs:range m1sop:FinancialHealthProfile ;
    rdfs:label "evaluates financial profile during call" .

m1sop:conductedByAgent rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:Call ;
    rdfs:range m1sop:CollectionAgent ;
    rdfs:label "conducted by agent" .

### 4. Workflow State Mapping
m1sop:currentMasterStage rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:Call ;
    rdfs:range m1sop:MasterSOPStage .

m1sop:currentSubStage rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:Call ;
    rdfs:range m1sop:SubStage .

m1sop:detectedRFD rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:Call ;
    rdfs:range m1sop:ReasonForDelinquency .

m1sop:loggedCompliance rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:Call ;
    rdfs:range m1sop:ComplianceArtifact .

m1sop:yieldedOutcome rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:Call ;
    rdfs:range m1sop:CallOutcome .

#################################################################
#    Data Properties
#################################################################

m1sop:accountNumber rdf:type owl:DatatypeProperty ;
    rdfs:domain m1sop:DebtAccount ;
    rdfs:range xsd:string .

m1sop:outstandingBalance rdf:type owl:DatatypeProperty ;
    rdfs:domain m1sop:DebtAccount ;
    rdfs:range xsd:decimal .

m1sop:isReceivingUIBenefits rdf:type owl:DatatypeProperty ;
    rdfs:domain m1sop:FinancialHealthProfile ;
    rdfs:range xsd:boolean .

m1sop:monthlyUIIncome rdf:type owl:DatatypeProperty ;
    rdfs:domain m1sop:FinancialHealthProfile ;
    rdfs:range xsd:decimal .

m1sop:severanceRemaining rdf:type owl:DatatypeProperty ;
    rdfs:domain m1sop:FinancialHealthProfile ;
    rdfs:range xsd:decimal .
