<http://example.org/ontology/m1-unemployment-sop> rdf:type owl:Ontology ;
    rdfs:label "Master 7-Stage SOP Ontology with M1-Unemployment Branching" ;
    rdfs:comment "Core ontology mapping master collection stages, RFD branching, domain entities, and dialogue state execution." .

#################################################################
#    Classes
#################################################################

### Master & Sub SOP Topology
m1sop:MasterSOPStage rdf:type owl:Class ;
    rdfs:label "Master SOP Stage" ;
    rdfs:comment "One of the 7 core stages in the overarching collection framework." .

m1sop:SubStage rdf:type owl:Class ;
    rdfs:label "Sub-Stage / Branch Stage" ;
    rdfs:comment "Specialized dialogue sub-states (e.g., M1 Unemployment sub-flow)." .

m1sop:ReasonForDelinquency rdf:type owl:Class ;
    rdfs:label "Reason For Delinquency (RFD)" ;
    rdfs:comment "Root cause of payment default." .

### Domain Entities
m1sop:Party rdf:type owl:Class .

m1sop:Debtor rdf:type owl:Class ;
    rdfs:subClassOf m1sop:Party ;
    rdfs:label "Debtor" .

m1sop:CollectionAgent rdf:type owl:Class ;
    rdfs:subClassOf m1sop:Party ;
    rdfs:label "Collection Agent" .

m1sop:DebtAccount rdf:type owl:Class ;
    rdfs:label "Debt Account" .

m1sop:Call rdf:type owl:Class ;
    rdfs:label "Call Interaction Session" .

m1sop:FinancialHealthProfile rdf:type owl:Class ;
    rdfs:label "Financial Health Profile" .

m1sop:ComplianceArtifact rdf:type owl:Class ;
    rdfs:label "Compliance Artifact" .

m1sop:CallOutcome rdf:type owl:Class ;
    rdfs:label "Call Outcome" .

m1sop:PromiseToPay rdf:type owl:Class ;
    rdfs:subClassOf m1sop:CallOutcome .

m1sop:ForbearanceAgreement rdf:type owl:Class ;
    rdfs:subClassOf m1sop:CallOutcome .

#################################################################
#    Object Properties
#################################################################

# State Transitions
m1sop:nextMasterStage rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:MasterSOPStage ;
    rdfs:range m1sop:MasterSOPStage .

m1sop:branchesToSubStage rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:MasterSOPStage ;
    rdfs:range m1sop:SubStage ;
    rdfs:comment "Links a master stage (Stage 3) to a specialized sub-flow." .

m1sop:nextSubStage rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:SubStage ;
    rdfs:range m1sop:SubStage .

m1sop:rejoinsMasterStage rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:SubStage ;
    rdfs:range m1sop:MasterSOPStage ;
    rdfs:comment "Routes a sub-flow back into the master pipeline (e.g., from profiling to Stage 4)." .

# Execution Links
m1sop:currentMasterStage rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:Call ;
    rdfs:range m1sop:MasterSOPStage .

m1sop:currentSubStage rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:Call ;
    rdfs:range m1sop:SubStage .

m1sop:detectedRFD rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:Call ;
    rdfs:range m1sop:ReasonForDelinquency .

m1sop:requiresCompliance rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:MasterSOPStage ;
    rdfs:range m1sop:ComplianceArtifact .

m1sop:loggedCompliance rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:Call ;
    rdfs:range m1sop:ComplianceArtifact .

m1sop:yieldedOutcome rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:Call ;
    rdfs:range m1sop:CallOutcome .

#################################################################
#    Individuals: Master 7-Stage Pipeline
#################################################################

m1sop:Stage_1_Opening rdf:type owl:NamedIndividual , m1sop:MasterSOPStage ;
    rdfs:label "1. Opening with Pre-Call Review & Account Initialization" ;
    m1sop:nextMasterStage m1sop:Stage_2_RPC .

m1sop:Stage_2_RPC rdf:type owl:NamedIndividual , m1sop:MasterSOPStage ;
    rdfs:label "2. Right Party Contact (RPC) & Verification" ;
    m1sop:requiresCompliance m1sop:Req_Verification , m1sop:Req_MiniMiranda ;
    m1sop:nextMasterStage m1sop:Stage_3_Inquiry .

m1sop:Stage_3_Inquiry rdf:type owl:NamedIndividual , m1sop:MasterSOPStage ;
    rdfs:label "3. Inquiry & Interaction (Branching Decision State)" ;
    m1sop:nextMasterStage m1sop:Stage_4_ApplyPressure ;
    m1sop:branchesToSubStage m1sop:SubStage_3_1_EmpathyPivot . # Conditional branch

m1sop:Stage_4_ApplyPressure rdf:type owl:NamedIndividual , m1sop:MasterSOPStage ;
    rdfs:label "4. Apply Pressure (Constructive Risk Realization)" ;
    m1sop:nextMasterStage m1sop:Stage_5_WorkoutSolution .

m1sop:Stage_5_WorkoutSolution rdf:type owl:NamedIndividual , m1sop:MasterSOPStage ;
    rdfs:label "5. Workout Solution" ;
    m1sop:nextMasterStage m1sop:Stage_6_PushPTP .

m1sop:Stage_6_PushPTP rdf:type owl:NamedIndividual , m1sop:MasterSOPStage ;
    rdfs:label "6. Push to Agree on Promise to Pay" ;
    m1sop:requiresCompliance m1sop:Req_ExplicitConsent ;
    m1sop:nextMasterStage m1sop:Stage_7_Closing .

m1sop:Stage_7_Closing rdf:type owl:NamedIndividual , m1sop:MasterSOPStage ;
    rdfs:label "7. Keep in Touch and Friendly Closing" .

#################################################################
#    Individuals: Stage 3 M1-Unemployment Sub-Flow
#################################################################

m1sop:RFD_InvoluntaryUnemployment rdf:type owl:NamedIndividual , m1sop:ReasonForDelinquency ;
    rdfs:label "Involuntary Unemployment" .

m1sop:SubStage_3_1_EmpathyPivot rdf:type owl:NamedIndividual , m1sop:SubStage ;
    rdfs:label "3.1 Empathic Pivot & De-escalation" ;
    m1sop:nextSubStage m1sop:SubStage_3_2_FinancialProfiling .

m1sop:SubStage_3_2_FinancialProfiling rdf:type owl:NamedIndividual , m1sop:SubStage ;
    rdfs:label "3.2 Unemployment Financial Profiling (UI & Severance Check)" ;
    m1sop:nextSubStage m1sop:SubStage_3_3_CardUtility .

m1sop:SubStage_3_3_CardUtility rdf:type owl:NamedIndividual , m1sop:SubStage ;
    rdfs:label "3.3 Card Utility & Line Assessment" ;
    m1sop:rejoinsMasterStage m1sop:Stage_4_ApplyPressure . # Returns to Master Pipeline

#################################################################
#    Individuals: Compliance Requirements

m1sop:Req_Verification rdf:type owl:NamedIndividual , m1sop:ComplianceArtifact ;
    rdfs:label "Right Party Identity Verified" .

m1sop:Req_MiniMiranda rdf:type owl:NamedIndividual , m1sop:ComplianceArtifact ;
    rdfs:label "Mini-Miranda Disclosure Delivered" .

m1sop:Req_ExplicitConsent rdf:type owl:NamedIndividual , m1sop:ComplianceArtifact ;
    rdfs:label "Explicit Verbal Consent Captured" .
