@prefix m1sop: <http://example.org/ontology/m1-unemployment-sop#> .
@prefix owl:   <http://www.w3.org/2002/07/owl#> .
@prefix rdf:   <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix xml:   <http://www.w3.org/XML/1998/namespace> .
@prefix xsd:   <http://www.w3.org/2001/XMLSchema#> .
@prefix rdfs:  <http://www.w3.org/2000/01/rdf-schema#> .
@base          <http://example.org/ontology/m1-unemployment-sop> .

<http://example.org/ontology/m1-unemployment-sop> rdf:type owl:Ontology ;
    rdfs:label "Extended M1-Unemployment Debt Recovery Domain & Dialogue Ontology" ;
    rdfs:comment "Comprehensive ontology modeling Debtors, Calls, Financial Profiles, Hardships, Promises-to-Pay, and the SOP execution state machine." .

#################################################################
#    Classes
#################################################################

### Core Actors & Account Entities

m1sop:Party rdf:type owl:Class ;
    rdfs:label "Party" ;
    rdfs:comment "Base class for participants in debt collection interactions." .

m1sop:Debtor rdf:type owl:Class ;
    rdfs:subClassOf m1sop:Party ;
    rdfs:label "Debtor" ;
    rdfs:comment "An individual owing an outstanding debt in early-stage (M1) delinquency." .

m1sop:CollectionAgent rdf:type owl:Class ;
    rdfs:subClassOf m1sop:Party ;
    rdfs:label "Collection Agent" ;
    rdfs:comment "Human collector or automated dialogue bot conducting the interaction." .

m1sop:DebtAccount rdf:type owl:Class ;
    rdfs:label "Debt Account" ;
    rdfs:comment "The financial obligation record subject to collection." .

### Interaction & Execution Entities

m1sop:Interaction rdf:type owl:Class ;
    rdfs:label "Interaction" ;
    rdfs:comment "An operational session or contact event." .

m1sop:Call rdf:type owl:Class ;
    rdfs:subClassOf m1sop:Interaction ;
    rdfs:label "Call" ;
    rdfs:comment "A specific telecommunication dialogue session between an Agent and a Debtor." .

### SOP & Dialogue Governance (State Machine)

m1sop:SOPPhase rdf:type owl:Class ;
    rdfs:label "SOP Phase" ;
    rdfs:comment "Macro phase of the call workflow." .

m1sop:SOPStage rdf:type owl:Class ;
    rdfs:label "SOP Stage" ;
    rdfs:comment "Micro step or dialogue state within an SOP Phase." .

m1sop:ComplianceArtifact rdf:type owl:Class ;
    rdfs:label "Compliance Artifact" ;
    rdfs:comment "Mandatory regulatory or audit proof logged during a call." .

### Domain Context: Hardship & Financials

m1sop:HardshipProfile rdf:type owl:Class ;
    rdfs:label "Hardship Profile" ;
    rdfs:comment "Reason for payment default." .

m1sop:InvoluntaryUnemploymentHardship rdf:type owl:Class ;
    rdfs:subClassOf m1sop:HardshipProfile ;
    rdfs:label "Involuntary Unemployment Hardship" ;
    rdfs:comment "Job loss due to layoff, business closure, or downsizing." .

m1sop:FinancialHealthProfile rdf:type owl:Class ;
    rdfs:label "Financial Health Profile" ;
    rdfs:comment "Assessment of current debtor liquidity and benefit streams." .

### Outcomes & Resolutions

m1sop:CallOutcome rdf:type owl:Class ;
    rdfs:label "Call Outcome" ;
    rdfs:comment "Formal resolution or commitment resulting from a Call." .

m1sop:PromiseToPay rdf:type owl:Class ;
    rdfs:subClassOf m1sop:CallOutcome ;
    rdfs:label "Promise To Pay (PTP)" ;
    rdfs:comment "Binding commitment to pay a specified amount on a future date." .

m1sop:MicroCommitment rdf:type owl:Class ;
    rdfs:subClassOf m1sop:CallOutcome ;
    rdfs:label "Micro-Commitment" ;
    rdfs:comment "Non-monetary agreement (e.g., agreeing to provide job status update next week)." .

m1sop:ForbearanceAgreement rdf:type owl:Class ;
    rdfs:subClassOf m1sop:CallOutcome ;
    rdfs:label "Forbearance Agreement" ;
    rdfs:comment "Temporary hold or restructuring granted due to unemployment." .

### Actor & Account Links

m1sop:hasAccount rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:Debtor ;
    rdfs:range m1sop:DebtAccount .

m1sop:conductedByAgent rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:Call ;
    rdfs:range m1sop:CollectionAgent .

m1sop:hasDebtorParticipant rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:Call ;
    rdfs:range m1sop:Debtor .

m1sop:concernsAccount rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:Call ;
    rdfs:range m1sop:DebtAccount .

### Hardship & Financial Links

m1sop:experiencesHardship rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:Debtor ;
    rdfs:range m1sop:HardshipProfile .

m1sop:evaluatedFinancials rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:Call ;
    rdfs:range m1sop:FinancialHealthProfile .

### Call Execution & SOP Progression Links

m1sop:currentSOPStage rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:Call ;
    rdfs:range m1sop:SOPStage ;
    rdfs:comment "The real-time stage of the ongoing call." .

m1sop:loggedCompliance rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:Call ;
    rdfs:range m1sop:ComplianceArtifact ;
    rdfs:comment "Compliance checkpoints satisfied during this specific call." .

m1sop:yieldedOutcome rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:Call ;
    rdfs:range m1sop:CallOutcome .

### SOP Topology (Flow Logic)

m1sop:containsStage rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:SOPPhase ;
    rdfs:range m1sop:SOPStage .

m1sop:nextStage rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:SOPStage ;
    rdfs:range m1sop:SOPStage .

m1sop:hasFallbackStage rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:SOPStage ;
    rdfs:range m1sop:SOPStage .

m1sop:requiresCompliance rdf:type owl:ObjectProperty ;
    rdfs:domain m1sop:SOPStage ;
    rdfs:range m1sop:ComplianceArtifact .
